   44: def create_pose_data_sample(keypoints_3d, frame_idx=0):
   45:     """Create PoseDataSample from 3D keypoints for visualization"""
   46:     # Extract coordinates and confidence
   47:     keypoints = np.array(keypoints_3d)  # [17, 4] -> x, y, z, conf
   48:     kpts_3d = keypoints[:, :3]  # [17, 3]
   49:     kpt_scores = keypoints[:, 3]  # [17]
   50:     
   51:     # Create instance data
   52:     pred_instances = InstanceData()
   53:     pred_instances.keypoints = kpts_3d.reshape(1, 17, 3)  # [1, 17, 3]
   54:     pred_instances.keypoint_scores = kpt_scores.reshape(1, 17)  # [1, 17]
   55:     
   56:     # Create pose data sample
   57:     pose_data_sample = PoseDataSample()
   58:     pose_data_sample.pred_instances = pred_instances
   59:     
   60:     return pose_data_sample

   63: def setup_visualizer():
   64:     """Setup 3D pose visualizer with COCO format"""
   65:     # Define colors for keypoints (BGR format)
   66:     kpt_color = [
   67:         [51, 153, 255], [51, 153, 255], [51, 153, 255], [51, 153, 255], [51, 153, 255],  # head
   68:         [0, 255, 0], [255, 128, 0], [0, 255, 0], [255, 128, 0], [0, 255, 0], [255, 128, 0],  # arms
   69:         [0, 255, 0], [255, 128, 0], [0, 255, 0], [255, 128, 0], [0, 255, 0], [255, 128, 0]  # legs
   70:     ]
   71:     
   72:     # Define colors for skeleton links
   73:     link_color = [
   74:         [0, 255, 0], [0, 255, 0], [255, 128, 0], [255, 128, 0], [51, 153, 255],  # legs
   75:         [51, 153, 255], [51, 153, 255], [51, 153, 255],  # torso
   76:         [0, 255, 0], [255, 128, 0], [0, 255, 0], [255, 128, 0],  # arms  
   77:         [51, 153, 255], [51, 153, 255], [51, 153, 255], [51, 153, 255], [51, 153, 255], [51, 153, 255], [51, 153, 255]  # head
   78:     ]
   79:     
   80:     visualizer = Pose3dLocalVisualizer(
   81:         name='azure_kinect_3d',
   82:         kpt_color=kpt_color,
   83:         link_color=link_color,
   84:         skeleton=COCO_SKELETON,
   85:         line_width=2,
   86:         radius=4
   87:     )
   88:     
   89:     # Set dataset meta info
   90:     dataset_meta = {
   91:         'dataset_name': 'azure_kinect_coco',
   92:         'keypoint_names': KEYPOINT_NAMES,
   93:         'skeleton_links': COCO_SKELETON,
   94:         'num_keypoints': 17
   95:     }
   96:     visualizer.set_dataset_meta(dataset_meta)
   97:     
   98:     return visualizer

  101: def _auto_axis_limit_from_persons(persons, kpt_thr=0.3, min_limit=1.5, scale=1.2):
  102:     """Compute a reasonable axis_limit from valid joints across persons."""
  103:     all_pts = []
  104:     for person in persons:
  105:         arr = np.asarray(person['keypoints_3d'], dtype=float)
  106:         conf = arr[:, 3] if arr.shape[1] >= 4 else np.ones(arr.shape[0])
  107:         mask = (conf > kpt_thr) & np.isfinite(arr[:, :3]).all(axis=1)
  108:         if np.any(mask):
  109:             all_pts.append(arr[mask, :3])
  110:     if not all_pts:
  111:         return min_limit
  112:     all_pts = np.concatenate(all_pts, axis=0)
  113:     ranges = np.ptp(all_pts, axis=0)
  114:     limit = max(min_limit, float(np.max(ranges) * scale))
  115:     return limit

  118: def visualize_single_frame(data, frame_idx=0, save_path=None, show=True,
  119:                            axis_limit=None, axis_dist=10.0,
  120:                            azim=70, elev=15, kpt_thr=0.3):
  121:     """Visualize single 3D pose frame"""
  122:     if frame_idx >= len(data['results']):
  123:         print(f"Frame {frame_idx} not found. Available frames: 0-{len(data['results'])-1}")
  124:         return
  125:     
  126:     frame_data = data['results'][frame_idx]
  127:     persons = frame_data.get('persons')
  128: 
  129:     # Setup visualizer
  130:     visualizer = setup_visualizer()
  131: 
  132:     # Create a blank image for visualization
  133:     img = np.ones((600, 800, 3), dtype=np.uint8) * 255
  134: 
  135:     # Auto axis limit if not provided
  136:     if axis_limit is None and 'persons' in frame_data:
  137:         axis_limit = _auto_axis_limit_from_persons(frame_data['persons'], kpt_thr)
  138:     if axis_limit is None:
  139:         axis_limit = 2.0
  140:     
  141:     # Draw 3D pose
  142:     if persons is None:
  143:         # Legacy single-person format: frame_data contains keypoints_3d directly
  144:         keypoints_3d = frame_data['keypoints_3d']
  145:         pose_sample = create_pose_data_sample(keypoints_3d, frame_idx)
  146:         visualizer._draw_3d_data_samples(
  147:             img,
  148:             pose_sample,
  149:             draw_gt=False,
  150:             kpt_thr=kpt_thr,
  151:             num_instances=1,
  152:             axis_azimuth=azim,
  153:             axis_elev=elev,
  154:             axis_limit=axis_limit,
  155:             axis_dist=axis_dist
  156:         )
  157:     else:
  158:         # Multi-person: draw each person sequentially
  159:         for p in persons:
  160:             keypoints_3d = p['keypoints_3d']
  161:             pose_sample = create_pose_data_sample(keypoints_3d, frame_idx)
  162:             visualizer._draw_3d_data_samples(
  163:                 img,
  164:                 pose_sample,
  165:                 draw_gt=False,
  166:                 kpt_thr=kpt_thr,
  167:                 num_instances=1,
  168:                 axis_azimuth=azim,
  169:                 axis_elev=elev,
  170:                 axis_limit=axis_limit,
  171:                 axis_dist=axis_dist
  172:             )
  173:     
  174:     # Get visualization result
  175:     vis_img = visualizer.get_image()
  176:     
  177:     if save_path:
  178:         mmcv.imwrite(vis_img, save_path)
  179:         print(f"Saved: {save_path}")
  180:     
  181:     if show:
  182:         cv2.imshow(f'3D Pose - Frame {frame_idx}', mmcv.rgb2bgr(vis_img))
  183:         cv2.waitKey(0)
  184:         cv2.destroyAllWindows()
  185:     
  186:     return vis_img

  189: def visualize_animation(data, save_dir=None, show=True, delay=500,
  190:                         axis_limit=None, axis_dist=10.0, azim=70, elev=15,
  191:                         kpt_thr=0.3):
  192:     """Create animation from multiple frames"""
  193:     if save_dir:
  194:         Path(save_dir).mkdir(exist_ok=True)
  195:     
  196:     visualizer = setup_visualizer()
  197:     
  198:     print(f"Creating animation from {len(data['results'])} frames...")
  199:     
  200:     for i, frame_data in enumerate(data['results']):
  201:         keypoints_3d = frame_data['keypoints_3d']
  202:         
  203:         # Create pose data sample
  204:         pose_sample = create_pose_data_sample(keypoints_3d, i)
  205:         
  206:         # Create image
  207:         img = np.ones((600, 800, 3), dtype=np.uint8) * 255
  208: 
  209:         if axis_limit is None and 'persons' in frame_data:
  210:             axis_limit_i = _auto_axis_limit_from_persons([{'keypoints_3d': keypoints_3d}], kpt_thr)
  211:         else:
  212:             axis_limit_i = axis_limit or 2.0
  213:         
  214:         # Draw 3D pose
  215:         visualizer._draw_3d_data_samples(
  216:             img,
  217:             pose_sample,
  218:             draw_gt=False,
  219:             kpt_thr=kpt_thr,
  220:             num_instances=1,
  221:             axis_azimuth=azim + i * 2,  # Rotate view slightly each frame
  222:             axis_elev=elev,
  223:             axis_limit=axis_limit_i,
  224:             axis_dist=axis_dist
  225:         )
  226:         
  227:         vis_img = visualizer.get_image()
  228:         
  229:         if save_dir:
  230:             save_path = os.path.join(save_dir, f"frame_{i:04d}.png")
  231:             mmcv.imwrite(vis_img, save_path)
  232:         
  233:         if show:
  234:             cv2.imshow('3D Pose Animation', mmcv.rgb2bgr(vis_img))
  235:             key = cv2.waitKey(delay)
  236:             if key == 27:  # ESC key
  237:                 break
  238:     
  239:     if show:
  240:         cv2.destroyAllWindows()
  241:     
  242:     print(f"Animation complete! Frames saved to: {save_dir}" if save_dir else "Animation complete!")

  245: def create_multi_view_comparison(data, frame_indices=None, save_path=None,
  246:                                  axis_limit=None, axis_dist=10.0,
  247:                                  azim=70, elev=15, kpt_thr=0.3):
  248:     """Create comparison view of multiple frames"""
  249:     if frame_indices is None:
  250:         frame_indices = list(range(min(4, len(data['results']))))
  251:     
  252:     n_frames = len(frame_indices)
  253:     cols = min(2, n_frames)
  254:     rows = (n_frames + cols - 1) // cols
  255:     
  256:     # Create combined image
  257:     img_h, img_w = 400, 600
  258:     combined_img = np.ones((img_h * rows, img_w * cols, 3), dtype=np.uint8) * 255
  259:     
  260:     visualizer = setup_visualizer()
  261:     
  262:     for i, frame_idx in enumerate(frame_indices):
  263:         if frame_idx >= len(data['results']):
  264:             continue
  265:             
  266:         frame_data = data['results'][frame_idx]
  267:         keypoints_3d = frame_data['keypoints_3d']
  268:         
  269:         # Create pose data sample
  270:         pose_sample = create_pose_data_sample(keypoints_3d, frame_idx)
  271:         
  272:         # Create individual image
  273:         img = np.ones((img_h, img_w, 3), dtype=np.uint8) * 255
  274: 
  275:         if axis_limit is None:
  276:             axis_limit_i = _auto_axis_limit_from_persons([{'keypoints_3d': keypoints_3d}], kpt_thr)
  277:         else:
  278:             axis_limit_i = axis_limit
  279:         
  280:         # Draw 3D pose
  281:         visualizer._draw_3d_data_samples(
  282:             img,
  283:             pose_sample,
  284:             draw_gt=False,
  285:             kpt_thr=kpt_thr,
  286:             num_instances=1,
  287:             axis_azimuth=azim + i * 30,
  288:             axis_elev=elev,
  289:             axis_limit=axis_limit_i,
  290:             axis_dist=axis_dist
  291:         )
  292:         
  293:         vis_img = visualizer.get_image()
  294:         
  295:         # Place in combined image
  296:         row = i // cols
  297:         col = i % cols
  298:         combined_img[row*img_h:(row+1)*img_h, col*img_w:(col+1)*img_w] = vis_img
  299:     
  300:     if save_path:
  301:         mmcv.imwrite(combined_img, save_path)
  302:         print(f"Multi-view saved: {save_path}")
  303:     
  304:     cv2.imshow('Multi-view 3D Poses', mmcv.rgb2bgr(combined_img))
  305:     cv2.waitKey(0)
  306:     cv2.destroyAllWindows()
  307:     
  308:     return combined_img
