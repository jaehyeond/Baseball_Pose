  112: def _apply_axis_transform(k3d, order='xyz', flip_z=False):
  113:     arr = np.asarray(k3d, dtype=float)
  114:     if arr.ndim != 2 or arr.shape[1] < 3:
  115:         return arr
  116:     idx_map = {
  117:         'xyz': (0, 1, 2),
  118:         'xzy': (0, 2, 1),
  119:         'yxz': (1, 0, 2),
  120:         'yzx': (1, 2, 0),
  121:         'zxy': (2, 0, 1),
  122:         'zyx': (2, 1, 0),
  123:     }
  124:     idx = idx_map.get(order, (0, 1, 2))
  125:     xyz = arr[:, idx].copy()
  126:     if flip_z:
  127:         xyz[:, 2] = -xyz[:, 2]
  128:     if arr.shape[1] > 3:
  129:         arr2 = np.concatenate([xyz, arr[:, 3:4]], axis=1)
  130:     else:
  131:         arr2 = xyz
  132:     return arr2

  135: def calculate_global_bounds(all_results, axis_order='xyz', flip_z=False, kpt_thr=0.3):
  136:     """Calculate global bounds across all frames for consistent visualization"""
  137:     all_points = []
  138:     
  139:     for frame_data in all_results:
  140:         persons = frame_data.get('persons', [])
  141:         for person in persons:
  142:             arr = np.array(person.get('keypoints_3d', person.get('joints_3d')))
  143:             arr = _apply_axis_transform(arr, order=axis_order, flip_z=flip_z)
  144:             scores = (arr[:, 3] if arr.shape[1] > 3 else np.ones((arr.shape[0],), dtype=float))
  145:             mask = (scores > kpt_thr) & np.isfinite(arr[:, :3]).all(axis=1)
  146:             if np.any(mask):
  147:                 all_points.append(arr[mask, :3])
  148:     
  149:     if len(all_points) > 0:
  150:         pts = np.concatenate(all_points, axis=0)
  151:         pmin = np.min(pts, axis=0)
  152:         pmax = np.max(pts, axis=0)
  153:         center_point = (pmin + pmax) / 2.0
  154:         # Í∏∞Ï°¥ ?∞Î™®?Ä ?ôÏùº???®Îî© Î∞©Ïãù ?ÅÏö©
  155:         max_range = max(pmax[0] - pmin[0], pmax[1] - pmin[1], pmax[2] - pmin[2])
  156:         if max_range < 1.0:
  157:             max_range = 1.0
  158:         padding = max_range * 0.2  # 20% ?®Îî© (?§Ïºà?àÌÜ§??Ï°∞Í∏à ???¨Í≤å)
  159:         range_meters = max_range + padding
  160:         return center_point, range_meters
  161:     else:
  162:         return [0, 0, 1.0], 2.0

  165: def assign_person_ids(persons, axis_order='xyz', flip_z=False, kpt_thr=0.3):
  166:     """Assign consistent IDs to persons based on position tracking"""
  167:     global person_tracker, next_person_id
  168:     
  169:     # Calculate center position for each person in current frame
  170:     current_persons = []
  171:     for i, person in enumerate(persons):
  172:         arr = np.array(person.get('keypoints_3d', person.get('joints_3d')))
  173:         arr = _apply_axis_transform(arr, order=axis_order, flip_z=flip_z)
  174:         scores = (arr[:, 3] if arr.shape[1] > 3 else np.ones((arr.shape[0],), dtype=float))
  175:         
  176:         # Find valid joints
  177:         valid_mask = (scores > kpt_thr) & np.isfinite(arr[:, :3]).all(axis=1)
  178:         if np.any(valid_mask):
  179:             valid_points = arr[valid_mask, :3]
  180:             center = np.mean(valid_points, axis=0)
  181:             current_persons.append({
  182:                 'index': i,
  183:                 'center': center,
  184:                 'person_data': person
  185:             })
  186:     
  187:     if not current_persons:
  188:         return []
  189:     
  190:     # If no previous tracking data, assign new IDs
  191:     if not person_tracker:
  192:         assigned_persons = []
  193:         for i, person_info in enumerate(current_persons):
  194:             person_id = next_person_id
  195:             next_person_id += 1
  196:             color = color_palette[i % len(color_palette)]
  197:             
  198:             person_tracker[person_id] = {
  199:                 'center': person_info['center'].copy(),
  200:                 'color': color
  201:             }
  202:             
  203:             assigned_persons.append({
  204:                 'id': person_id,
  205:                 'color': color,
  206:                 'data': person_info['person_data']
  207:             })
  208:         return assigned_persons
  209:     
  210:     # Match current persons to previous ones based on distance
  211:     current_centers = np.array([p['center'] for p in current_persons])
  212:     previous_centers = np.array([person_tracker[pid]['center'] for pid in person_tracker.keys()])
  213:     previous_ids = list(person_tracker.keys())
  214:     
  215:     # Calculate distance matrix
  216:     if len(previous_centers) > 0:
  217:         distances = cdist(current_centers, previous_centers)
  218:         
  219:         # Hungarian algorithm-like assignment (simple greedy for now)
  220:         assigned_persons = []
  221:         used_ids = set()
  222:         
  223:         # Sort current persons by minimum distance to any previous person
  224:         person_indices = list(range(len(current_persons)))
  225:         person_indices.sort(key=lambda i: np.min(distances[i]))
  226:         
  227:         for curr_idx in person_indices:
  228:             person_info = current_persons[curr_idx]
  229:             
  230:             # Find closest previous person that hasn't been assigned
  231:             available_ids = [i for i, pid in enumerate(previous_ids) if pid not in used_ids]
  232:             if available_ids:
  233:                 closest_prev_idx = available_ids[np.argmin(distances[curr_idx, available_ids])]
  234:                 person_id = previous_ids[closest_prev_idx]
  235:                 used_ids.add(person_id)
  236:                 
  237:                 # Update tracker with new position
  238:                 person_tracker[person_id]['center'] = person_info['center'].copy()
  239:                 
  240:                 assigned_persons.append({
  241:                     'id': person_id,
  242:                     'color': person_tracker[person_id]['color'],
  243:                     'data': person_info['person_data']
  244:                 })
  245:             else:
  246:                 # New person - assign new ID
  247:                 person_id = next_person_id
  248:                 next_person_id += 1
  249:                 # Use next available color
  250:                 used_colors = [person_tracker[pid]['color'] for pid in used_ids]
  251:                 available_colors = [c for c in color_palette if c not in used_colors]
  252:                 color = available_colors[0] if available_colors else color_palette[person_id % len(color_palette)]
  253:                 
  254:                 person_tracker[person_id] = {
  255:                     'center': person_info['center'].copy(),
  256:                     'color': color
  257:                 }
  258:                 
  259:                 assigned_persons.append({
  260:                     'id': person_id,
  261:                     'color': color,
  262:                     'data': person_info['person_data']
  263:                 })
  264:         
  265:         return assigned_persons
  266:     else:
  267:         # Fresh start
  268:         return assign_person_ids(persons, axis_order, flip_z, kpt_thr)
